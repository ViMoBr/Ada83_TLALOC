Contexte : le crash T2s est dans la grosse ligne SONDEA ; ingredients
non prouves restants : egalite TREE variable/variable compilee au site,
BOOLEAN'IMAGE sur valeur calculee, caténation a 8 operandes. Ce lot
eclate SONDEA en pieces atomiques (A1..A6) et convertit SONDEB1/B2/B3
aux memes primitives, pour (i) designer l'ingredient tueur et (ii)
donner aux sondes une chance de SURVIVRE jusqu'au littéral RED et de
montrer la divergence originelle en direct.

------------------------------------------------------------------------
MODIFICATION 15 — eclatement de SONDEA en A1..A6
------------------------------------------------------------------------
ANCRE (unique, inchangee) :
	Put_Line ("SONDE2E2");

SUPPRIMER (ligne unique — la grosse sonde SONDEA de la modification 1) :
	Put_Line ("SONDEA " & PRINT_NAME (D (LX_SYMREP, DESIGNATOR)) & " CAND=" & NODE_NAME'IMAGE (TEMP.TY) & " REG=" & BOOLEAN'IMAGE (D (XD_REGION_DEF, TEMP) = NAME_DEF) & " SPEC=" & BOOLEAN'IMAGE (DB (XD_IS_IN_SPEC, TEMP)));

REMPLACER PAR :
	Put_Line ("SONDEA1 " & PRINT_NAME (D (LX_SYMREP, DESIGNATOR)));
	Put_Line ("SONDEA2 " & NODE_NAME'IMAGE (TEMP.TY));
	if D (XD_REGION_DEF, TEMP) = NAME_DEF then
	  Put_Line ("SONDEA3 REG=TRUE");
	else
	  Put_Line ("SONDEA3 REG=FALSE");
	end if;
	if DB (XD_IS_IN_SPEC, TEMP) then
	  Put_Line ("SONDEA4 SPEC=TRUE");
	else
	  Put_Line ("SONDEA4 SPEC=FALSE");
	end if;
	Put_Line ("SONDEA5 " & BOOLEAN'IMAGE (D (XD_REGION_DEF, TEMP) = NAME_DEF));
	Put_Line ("SONDEA6 " & "A" & "B" & "C" & "D" & "E" & "F" & "G");

(A1/A2 : formes prouvees ; A3 : egalite TREE nue, sans image ni
caténation ; A4 : DB nu ; A5 : BOOLEAN'IMAGE sur valeur calculee,
atteint seulement si A3/A4 passent ; A6 : caténation longue de
litteraux purs, sans appel.)

------------------------------------------------------------------------
MODIFICATION 16 — SONDEB1 en primitives
------------------------------------------------------------------------
ANCRE (unique, inchangee) :
	OLD_ID := D (XD_SOURCE_NAME, GET_DEF (OLD_DEFINTERP));

SUPPRIMER (ligne unique — la sonde B1 de la modification 2) :
	Put_Line ("SONDEB1 " & NODE_NAME'IMAGE (OLD_ID.TY) & " OP=" & BOOLEAN'IMAGE (OLD_ID.TY = DN_BLTN_OPERATOR_ID) & " LIT=" & BOOLEAN'IMAGE (OLD_ID.TY in CLASS_ENUM_LITERAL) & " INS=" & BOOLEAN'IMAGE (OLD_ID.TY in CLASS_SUBPROG_NAME and then D (SM_UNIT_DESC, OLD_ID).TY = DN_INSTANTIATION));

REMPLACER PAR :
	Put_Line ("SONDEB1 " & NODE_NAME'IMAGE (OLD_ID.TY));
	if OLD_ID.TY = DN_BLTN_OPERATOR_ID then
	  Put_Line ("SONDEB1-OP");
	end if;
	if OLD_ID.TY in CLASS_ENUM_LITERAL then
	  Put_Line ("SONDEB1-LIT");
	end if;
	if OLD_ID.TY in CLASS_SUBPROG_NAME and then D (SM_UNIT_DESC, OLD_ID).TY = DN_INSTANTIATION then
	  Put_Line ("SONDEB1-INS");
	end if;

de toute facon quand NE=TRUE, et B3 en montre l'effet.)
