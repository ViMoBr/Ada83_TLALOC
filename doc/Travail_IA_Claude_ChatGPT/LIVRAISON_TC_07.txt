=======================================================================
LIVRAISON TC-07 — TARGET_CODE : protocole d'appel complet
  (CALL, CALLI, RTD, LSPA ; PRO/ELB/endPRO sous le cmp)
=======================================================================

COMMIT 1 — "EMITS : CALL rel32 vers .elab (lazy reel), CALLI, RTD,
            LSPA canonique ; PRO = BRA post, ELB = elab + LINK
            loc_siz, macros de declaration en taille zero ; temoin
            appel inter-frames"

SEMANTIQUE (relue au codi) :
  - CALL prefix,subname : E8 rel32 vers prefix#subname#_ = adresse de
    prefix.subname.elab (5 octets FIXES). TARGET_CODE resout
    directement le chemin .elab (point final du prefixe gere) — les
    marques LAZY restent booleennes, equivalent strict ;
  - LSPA : movabs de la meme adresse + PUSH (18, forme canonique) ;
  - CALLI : POP_RAX + FF D0 (10) ;
  - RTD p : desallocation lea rbp,-p8/-p32 puis C3 (1 / 5 / 8) ;
  - PRO : BRA post (5) — contournement d'elaboration ;
  - ELB lvl : label elab ICI puis LINK lvl, loc_siz (taille = LINK) ;
  - endPRO : label post ICI, zero octet ;
  - PRMS, PRM, endPRMS, VAR, STATOFS : zero octet (pure declaration).

ORACLE DU COMMIT :
  1) septuple PASSE ("PASSE call"), aucune ligne ECHEC — ASM_SIZE =
     403 (82 + 321, compte a la main), elab de SP7 a ENTRY+245,
     post a ENTRY+403 ;
  2) fasmg TC_TEST7.FAS TC_REF7 && cmp TC_REF7 TC_TEST7.BIN ->
     AUCUNE difference (premier PRO complet, lazy des deux cotes :
     multipasse fasmg contre atteignabilite P1 — MEMES OCTETS) ;
  3) ./TC_TEST7.BIN ; echo $? -> 0 (1 = la valeur 42 n'est pas
     revenue dans la frame de main via le display).

  function		ELAB_TARGET ( E :IR.ELT_ID ) return SYMBOLS.VALUE_TYPE
  --| adresse de prefix.subname.elab pour CALL/LSPA — le prefixe des FINC
  --| reels porte un POINT FINAL de concatenation (releve EXP-03) ; le
  --| temoin, lui, ecrit "STANDARD" nu : les deux formes sont acceptees.
  is
    P			: constant STRING := LEX.IMAGE( IR.OP_TXT( E, 1 ) );
    S			: constant STRING := LEX.IMAGE( IR.OP_TXT( E, 2 ) );
  begin
    if IR.N_OPS( E ) < 2
       or else  IR.OP_TAG( E, 1 ) /= IR.NAME_OP
       or else  IR.OP_TAG( E, 2 ) /= IR.NAME_OP
    then
      FAULT( "CALL/LSPA : prefixe, sous-nom attendus" );
    end if;
    if P( P'LAST ) = '.'
    then
      return SYMBOLS.VALUE_OF( SYMBOLS.RESOLVE( P & S & ".elab" ) );
    end if;
    return SYMBOLS.VALUE_OF( SYMBOLS.RESOLVE( P & "." & S & ".elab" ) );
  end ELAB_TARGET;

  --|  TEMOIN TEMPORAIRE APPEL (TC-07) — CALL vers .elab (lazy reel des
  --|  deux cotes : multipasse fasmg / atteignabilite P1), PRO/ELB/endPRO,
  --|  RTD, ecriture inter-frames par le display. Verdict du binaire :
  --|  0 = 42 revenu dans la frame de main, 1 = non. A RETIRER.
  declare
    use LEX;
    use SYMBOLS;
    use IR;
    OK			: BOOLEAN	:= TRUE;
    F			: FILE_TYPE;
    FROM7		: IR.ELT_ID;

    procedure CHECK ( COND :BOOLEAN; MSG :STRING )
    is
    begin
      if not COND
      then
	OK := FALSE;
	PUT_LINE( "ECHEC call : " & MSG );
      end if;
    end CHECK;

  begin
    CREATE( F, OUT_FILE, "TC_TEST7.FAS" );
    PUT_LINE( F, "include 'codi_x86_64.finc'" );
    PUT_LINE( F, "namespace STANDARD" );
    PUT_LINE( F, "main7:" );
    PUT_LINE( F, "	LINK	1, 32" );
    PUT_LINE( F, "	CALL	STANDARD, SP7" );
    PUT_LINE( F, "	Ld	1, 8" );
    PUT_LINE( F, "	LI	42" );
    PUT_LINE( F, "	CEQ" );
    PUT_LINE( F, "	BT	ok7" );
    PUT_LINE( F, "	UNLINK	1" );
    PUT_LINE( F, "	SYS_EXIT	1" );
    PUT_LINE( F, "ok7:" );
    PUT_LINE( F, "	UNLINK	1" );
    PUT_LINE( F, "	SYS_EXIT	0" );
    PUT_LINE( F, "if defined SP7_" );
    PUT_LINE( F, "	PRO	SP7" );
    PUT_LINE( F, "	ELB	2" );
    PUT_LINE( F, "	VAR	T7_disp, q" );
    PUT_LINE( F, "	LI	6" );
    PUT_LINE( F, "	LI	7" );
    PUT_LINE( F, "	MUL" );
    PUT_LINE( F, "	Sq	2, T7_disp" );
    PUT_LINE( F, "	Lq	2, T7_disp" );
    PUT_LINE( F, "	Sd	1, 8" );
    PUT_LINE( F, "	UNLINK	2" );
    PUT_LINE( F, "	RTD	0" );
    PUT_LINE( F, "	endPRO" );
    PUT_LINE( F, "end if" );
    PUT_LINE( F, "end namespace" );
    CLOSE( F );

    FROM7 := IR.ELT_ID( IR.ELT_COUNT + 1 );
    RUN_P0( "TC_TEST7.FAS" );

    CHECK( VALUE_OF( RESOLVE( "STANDARD.SP7.post" ) ) = 16#400078# + 403,
	   "post de SP7 en fin de corps" );

      PUT_LINE( "  chmod +x TC_TEST7.BIN && ./TC_TEST7.BIN ; echo $?   (attendu : 0)" );
