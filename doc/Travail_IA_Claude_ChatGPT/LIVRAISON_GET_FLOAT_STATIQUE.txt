=========================================================================
COMMIT B -- text_io.adb, GET(STRING) : MAX_LIMBS statique + sticky
=========================================================================
Fichier : text_io.adb (procedure GET sur STRING de FLOAT_IO)

Motif : borne de composant record statique (deblocage expander) ; la
correction d'arrondi reste exacte pour toute longueur de chaine via la
troncature a 800 chiffres significatifs + bit sticky (borne de Gay :
768 chiffres suffisent pour binary64).

--- MODIF B1 : MAX_LIMBS statique, SIG_DIGITS_MAX ----------------------
ANCRE (unique, inchangee) :
      BASE : constant INTEGER := 32768;       -- 2**15

SUPPRIMER :
      -- Le +384 fournit largement la place necessaire aux decalages
      -- binaires IEEE binary64 (jusqu'a 1074 bits) ainsi qu'aux
      -- puissances de dix.
      MAX_LIMBS : constant POSITIVE
        := ( FROM'LENGTH + 3 ) / 4 + 384;

REMPLACER PAR :
      -- STATIQUE (nombre nomme) : le composant D du record BIG est
      -- ainsi contraint statiquement -- offsets record statiques.
      --
      -- Dimensionnement, au pire cas passant les gardes DEC_TOP :
      --   SIG <= 10**SIG_DIGITS_MAX                    ~ 2659 bits
      --   A   <= 10**1025                              ~ 3406 bits
      --   B   <= 10**(1074 + SIG_DIGITS_MAX)           ~ 6227 bits
      --   decalages 2**K de la localisation de E       ~ 6240 bits
      -- soit au plus ~420 limbs de 15 bits ; 600 laisse une marge.
      MAX_LIMBS : constant := 600;

      -- Nombre maximal de chiffres significatifs absorbes dans SIG.
      -- 768 chiffres suffisent pour arrondir correctement binary64
      -- (borne de Gay) ; les chiffres excedentaires ne comptent que
      -- via DROPPED et STICKY ci-dessous. 800 donne une marge.
      SIG_DIGITS_MAX : constant := 800;

--- MODIF B2 : variables DROPPED et STICKY ------------------------------
ANCRE (unique, inchangee) :
      NONZERO_SEEN    : BOOLEAN := FALSE;

SUPPRIMER (2 lignes sous l'ancre, apres la ligne vide) :
      FRAC_COUNT      : INTEGER := 0;
      SIG_COUNT       : INTEGER := 0;

REMPLACER PAR :
      FRAC_COUNT      : INTEGER := 0;
      SIG_COUNT       : INTEGER := 0;

      -- Chiffres significatifs au-dela de SIG_DIGITS_MAX, non absorbes
      -- dans SIG :
      -- DROPPED : ceux situes avant le point (facteur 10 chacun) ;
      -- STICKY  : vrai si l'un au moins etait non nul -- la valeur
      --           exacte est alors STRICTEMENT superieure a la valeur
      --           tronquee representee par SIG.
      DROPPED         : INTEGER := 0;
      STICKY          : BOOLEAN := FALSE;

--- MODIF B3 : branche chiffre de la mantisse ---------------------------
ANCRE (unique, inchangee) :
      ----------------------------------------------------------------
      -- Mantisse
      ----------------------------------------------------------------

SUPPRIMER (dans la boucle qui suit l'ancre) :
        if CH >= '0' and then CH <= '9' then

          DIG := CHARACTER'POS(CH) - CHARACTER'POS('0');

          MUL_SMALL( SIG, 10 );
          ADD_SMALL( SIG, DIG );

          HAVE_DIGIT := TRUE;

          if NONZERO_SEEN then
            SIG_COUNT := SIG_COUNT + 1;

          elsif DIG /= 0 then
            NONZERO_SEEN := TRUE;
            SIG_COUNT    := 1;
          end if;

          if POINT_SEEN then
            FRAC_COUNT := FRAC_COUNT + 1;
          end if;

          LAST := POS;
          POS  := POS + 1;

REMPLACER PAR :
        if CH >= '0' and then CH <= '9' then

          DIG := CHARACTER'POS(CH) - CHARACTER'POS('0');

          if SIG_COUNT < SIG_DIGITS_MAX then

            MUL_SMALL( SIG, 10 );
            ADD_SMALL( SIG, DIG );

            if NONZERO_SEEN then
              SIG_COUNT := SIG_COUNT + 1;

            elsif DIG /= 0 then
              NONZERO_SEEN := TRUE;
              SIG_COUNT    := 1;
            end if;

            if POINT_SEEN then
              FRAC_COUNT := FRAC_COUNT + 1;
            end if;

          else

            -- Chiffre au-dela de la capacite : il ne modifie ni SIG
            -- ni FRAC_COUNT. Avant le point il vaut un facteur 10 ;
            -- partout il alimente le bit sticky.

            if not POINT_SEEN then
              DROPPED := DROPPED + 1;
            end if;

            if DIG /= 0 then
              STICKY := TRUE;
            end if;

          end if;

          HAVE_DIGIT := TRUE;

          LAST := POS;
          POS  := POS + 1;

--- MODIF B4 : DEC_EXP integre DROPPED ----------------------------------
ANCRE (unique, inchangee) :
      --   SIG * 10**DEC_EXP

SUPPRIMER :
      if EXP_NEG then
        DEC_EXP := -EXP_ABS - FRAC_COUNT;
      else
        DEC_EXP :=  EXP_ABS - FRAC_COUNT;
      end if;

REMPLACER PAR :
      if EXP_NEG then
        DEC_EXP := -EXP_ABS - FRAC_COUNT + DROPPED;
      else
        DEC_EXP :=  EXP_ABS - FRAC_COUNT + DROPPED;
      end if;

--- MODIF B6 : arrondi, cas C = 0 avec sticky ---------------------------
ANCRE (unique, inchangee) :

        elsif C = 0 then
