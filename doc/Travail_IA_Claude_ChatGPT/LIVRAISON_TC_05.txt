LIVRAISON TC-05 — TARGET_CODE : constantes differees LIFO (STR/CST),
  LCA + SYS_PUT_STR, et trois corrections de conformite au codi
=======================================================================

COMMIT 1 — "differes LIFO (DIS_BONJOUR executable), LCA/SYS_PUT_STR ;
            USEINFO vraie signature, STR = namespace, point final des
            prefixes CALL"

SEMANTIQUES RELUES (NOTE v2 a mettre a jour) :
  - STR name,'bytes' cree un NAMESPACE name contenant data_ptr (dq),
    info_ptr (dq), info (dd bits,8,1,len), data (les octets), et les
    statiques SIZ/COMP_SIZ/FST_1/LST_1 = 0/4/8/12 ; le tout DIFFERE,
    place LIFO apres le code, aligne qword, bourrage 16#90# ;
  - CST name,c,val : align_c, name = $, d-c val ; differe LIFO aussi ;
  - USEINFO lvl, field_name, load_instruc& : TROIS operandes — mon P2
    lisait le nom en op1 (le temoin TC-03, coherent avec ma faute,
    passait a tort) ;
  - le prefixe des CALL/LSPA porte un POINT FINAL de concatenation
    (releve EXP-03) : MARK doit le depouiller ;
  - SYS_PUT_STR : POP_RSI puis 23 octets (31 au total), consomme
    l'adresse du doublet (= name.data_ptr).

ORACLE DU COMMIT :
  1) quintuple PASSE (symbols, lex, passes, emits, defer), aucune
     ligne ECHEC — le temoin interne verifie notamment le LIFO :
     la DERNIERE STR enregistree recoit l'adresse la plus BASSE ;
  2) byte-diff : fasmg TC_TEST5.FAS TC_REF5 && cmp TC_REF5
     TC_TEST5.BIN -> AUCUNE difference (bourrage 16#90# compris) ;
  3) EXECUTION : ./TC_TEST5.BIN affiche exactement "okAH" et sort
     en 0 (idem TC_REF5) — premiere sortie visible d'un binaire
     assemble nativement ;
  4) le temoin TC-03 reste vert avec la vraie signature USEINFO.

  procedure		E_POP_RSI					--| 8 octets (SYS_PUT_STR)
  is
  begin
    B( 16#48# ); B( 16#8B# ); B( 16#75# ); B( 16#00# );			--| mov rsi, [rbp]
    B( 16#48# ); B( 16#8D# ); B( 16#6D# ); B( 16#F8# );			--| lea rbp, [rbp-8]
  end E_POP_RSI;

  function		ALIGNED8 ( P :SYMBOLS.VALUE_TYPE ) return SYMBOLS.VALUE_TYPE
  is
  begin
    if P mod 8 = 0
    then
      return P;
    end if;
    return P + 8 - ( P mod 8 );
  end ALIGNED8;

  function		STRB_LEN ( S :LEX.SLICE ) return SYMBOLS.VALUE_TYPE
  --| longueur EFFECTIVE d'une tranche STRB : les '' doubles comptent 1
  is
    N			: SYMBOLS.VALUE_TYPE := 0;
    I			: NATURAL	:= S.F;
    T			: constant STRING := LEX.IMAGE( S );
  begin
    I := T'FIRST;
    while I <= T'LAST
    loop
      if T( I ) = '''  and then  I < T'LAST  and then  T( I + 1 ) = '''
      then
	I := I + 1;							--| doublage : un seul octet
      end if;
      N := N + 1;
      I := I + 1;
    end loop;
    return N;
  end STRB_LEN;

  --|  Differes STR/CST : UNE routine, deux modes (LIFO : dernier
  --|  enregistre place/emis en PREMIER — releve DIS_BONJOUR).
  --|  PLACING : pose les adresses (P2B) ; sinon : emet (P3), bourrage
  --|  16#90#, et VERIFIE que chaque adresse emise egale l'adresse posee.
  procedure		DO_DEFERRED ( PLACING :BOOLEAN;
				      CUR :in out SYMBOLS.VALUE_TYPE )
  is
    S0			: constant SYMBOLS.SCOPE_ID := SYMBOLS.CURRENT_SCOPE;
  begin
    for DI in reverse 1 .. IR.DEFER_COUNT
    loop
      declare
	E		: constant IR.ELT_ID := IR.DEFER_AT( DI );
	M		: constant STRING    := LEX.IMAGE( IR.MNEMO_OF( E ) );
	NAME		: constant STRING    := LEX.IMAGE( IR.OP_TXT( E, 1 ) );
      begin
	SYMBOLS.USE_SCOPE( IR.SCOPE_OF( E ) );
	if M = "STR"
	then
	  declare
	    LEN		: constant SYMBOLS.VALUE_TYPE := STRB_LEN( IR.OP_TXT( E, 2 ) );
	    P		: constant SYMBOLS.VALUE_TYPE := ALIGNED8( CUR );
	    T		: constant STRING := LEX.IMAGE( IR.OP_TXT( E, 2 ) );
	    I		: NATURAL;
	  begin
	    if PLACING
	    then
	      SYMBOLS.SET_VALUE( SYMBOLS.RESOLVE( NAME & ".data_ptr" ), P );
	      SYMBOLS.SET_VALUE( SYMBOLS.RESOLVE( NAME & ".info_ptr" ), P + 8 );
	      SYMBOLS.SET_VALUE( SYMBOLS.RESOLVE( NAME & ".info" ),     P + 16 );
	      SYMBOLS.SET_VALUE( SYMBOLS.RESOLVE( NAME & ".data" ),     P + 32 );
	    else
	      while ORG + SYMBOLS.VALUE_TYPE( TOP ) < P
	      loop
		B( 16#90# );						--| bourrage align_q (byte-diff !)
	      end loop;
	      if SYMBOLS.VALUE_OF( SYMBOLS.RESOLVE( NAME & ".data_ptr" ) )
		 /= ORG + SYMBOLS.VALUE_TYPE( TOP )
	      then
		FAULT( "placement STR incoherent P2B/P3 : " & NAME );
	      end if;
	      Q64( P + 32 );						--| dq data
	      Q64( P + 16 );						--| dq info
	      D32( INTEGER( 8 * LEN ) );				--| SIZ  (bits)
	      D32( 8 );							--| COMP_SIZ
	      D32( 1 );							--| FST_1
	      D32( INTEGER( LEN ) );					--| LST_1
	      I := T'FIRST;
	      while I <= T'LAST
	      loop
		if T( I ) = '''  and then  I < T'LAST  and then  T( I + 1 ) = '''
		then
		  I := I + 1;						--| '' -> un seul '
		end if;
		B( CHARACTER'POS( T( I ) ) );
		I := I + 1;
	      end loop;
	    end if;
	    CUR := P + 32 + LEN;
	  end;

	elsif M = "CST"
	then
	  declare
	    C		: constant STRING := LEX.IMAGE( IR.OP_TXT( E, 2 ) );
	    UNIT	: SYMBOLS.VALUE_TYPE;
	    P		: SYMBOLS.VALUE_TYPE;
	    VAL		: SYMBOLS.VALUE_TYPE;
	  begin
	    if C = "b"
	    then
	      UNIT := 1;
	    elsif C = "w"
	    then
	      UNIT := 2;
	    elsif C = "d"
	    then
	      UNIT := 4;
	    elsif C = "q"
	    then
	      UNIT := 8;
	    else
	      FAULT( "CST : unite inconnue : " & C );
	    end if;
	    if IR.OP_TAG( E, 3 ) /= IR.INT_OP
	    then
	      FAULT( "CST : valeur litterale attendue (tranche TC-05)" );
	    end if;
	    VAL := IR.OP_INT( E, 3 );
	    P := CUR;
	    if UNIT > 1  and then  P mod UNIT /= 0
	    then
	      P := P + UNIT - ( P mod UNIT );
	    end if;
	    if PLACING
	    then
	      SYMBOLS.SET_VALUE( SYMBOLS.RESOLVE( NAME ), P );
	    else
	      while ORG + SYMBOLS.VALUE_TYPE( TOP ) < P
	      loop
		B( 16#90# );
	      end loop;
	      if UNIT = 1
	      then
		B( INTEGER( VAL mod 256 ) );
	      elsif UNIT = 2
	      then
		W16( INTEGER( VAL mod 65536 ) );
	      elsif UNIT = 4
	      then
		D32( INTEGER( VAL mod 4294967296 ) );
	      else
		Q64( VAL );
	      end if;
	    end if;
	    CUR := P + UNIT;
	  end;

	else
	  FAULT( "differe inattendu : " & M );
	end if;
      end;
    end loop;
    SYMBOLS.USE_SCOPE( S0 );
  end DO_DEFERRED;

  --|  TEMOIN TEMPORAIRE DES DIFFERES (TC-05) — LIFO, LCA, SYS_PUT_STR.
  --|  Sortie attendue du binaire : "okAH". A RETIRER avec les autres.
  declare
    use LEX;
    use SYMBOLS;
    use IR;
    OK			: BOOLEAN	:= TRUE;
    F			: FILE_TYPE;
    FROM5		: IR.ELT_ID;

    procedure CHECK ( COND :BOOLEAN; MSG :STRING )
    is
    begin
      if not COND
      then
	OK := FALSE;
	PUT_LINE( "ECHEC defer : " & MSG );
      end if;
    end CHECK;

  begin
    CREATE( F, OUT_FILE, "TC_TEST5.FAS" );
    PUT_LINE( F, "include 'codi_x86_64.finc'" );
    PUT_LINE( F, "namespace STANDARD" );
    PUT_LINE( F, "main5:" );
    PUT_LINE( F, "	LCA	MSG5A.data_ptr" );
    PUT_LINE( F, "	SYS_PUT_STR" );
    PUT_LINE( F, "	LCA	MSG5B.data_ptr" );
    PUT_LINE( F, "	SYS_PUT_STR" );
    PUT_LINE( F, "	SYS_EXIT	0" );
    PUT_LINE( F, "STR	MSG5A, 'ok'" );
    PUT_LINE( F, "STR	MSG5B, 'AH'" );
    PUT_LINE( F, "	CST	K5, q, 123" );
    PUT_LINE( F, "end namespace" );
    CLOSE( F );

    FROM5 := IR.ELT_ID( IR.ELT_COUNT + 1 );
    RUN_P0( "TC_TEST5.FAS" );
    EMITS.P2B_ADDRESSES( FROM5, IR.ELT_ID( IR.ELT_COUNT ) );
    EMITS.P3_EMIT( FROM5, IR.ELT_ID( IR.ELT_COUNT ), "TC_TEST5.BIN" );

    --  LIFO (DIS_BONJOUR) : dernier enregistre = premier place.
    --  Ordre d'enregistrement : MSG5A, MSG5B, K5 -> placement K5,
    --  MSG5B, MSG5A ; donc K5 < MSG5B < MSG5A en adresses.
    CHECK( VALUE_OF( RESOLVE( "STANDARD.K5" ) )
	     < VALUE_OF( RESOLVE( "STANDARD.MSG5B.data_ptr" ) )
	   and then VALUE_OF( RESOLVE( "STANDARD.MSG5B.data_ptr" ) )
	     < VALUE_OF( RESOLVE( "STANDARD.MSG5A.data_ptr" ) ),
	   "placement LIFO des differes" );
    CHECK( VALUE_OF( RESOLVE( "STANDARD.MSG5A.data" ) )
	     = VALUE_OF( RESOLVE( "STANDARD.MSG5A.data_ptr" ) ) + 32,
	   "structure du bloc STR (data a +32)" );
    CHECK( VALUE_OF( RESOLVE( "STANDARD.MSG5A.SIZ" ) ) = 0
	   and then VALUE_OF( RESOLVE( "STANDARD.MSG5A.LST_1" ) ) = 12,
	   "statiques du STR (0/4/8/12)" );

    if OK
    then
      PUT_LINE( "PASSE defer" );
      PUT_LINE( "TC_TEST5.BIN ecrit — oracle externe :" );
      PUT_LINE( "  fasmg TC_TEST5.FAS TC_REF5 && cmp TC_REF5 TC_TEST5.BIN" );
      PUT_LINE( "  chmod +x TC_TEST5.BIN && ./TC_TEST5.BIN ; echo $?   (attendu : okAH puis 0)" );
    end if;
  end;
